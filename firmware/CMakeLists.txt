cmake_minimum_required(VERSION 3.20)
project(explorer_controller VERSION 0.2.2 LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
add_library(explorer_core core/src/base.c core/src/servo.c core/src/profile.c core/src/wire.c core/src/controller.c)
target_include_directories(explorer_core PUBLIC core/include)
target_compile_options(explorer_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror -fno-fast-math)
if(NOT CMAKE_CROSSCOMPILING)
  option(EX_SANITIZE "Address/undefined sanitizers" ON)
  if(EX_SANITIZE)
    target_compile_options(explorer_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(explorer_core PUBLIC -fsanitize=address,undefined)
  endif()
  enable_testing()
  foreach(name core runtime)
    add_executable(${name}_tests core/tests/test_${name}.c)
    target_link_libraries(${name}_tests PRIVATE explorer_core m)
    add_test(NAME ${name} COMMAND ${name}_tests)
  endforeach()
  add_executable(servo_bus_tests tests/test_servo_bus.c src/servo_bus.c)
  target_include_directories(servo_bus_tests BEFORE PRIVATE tests/mock include)
  target_link_libraries(servo_bus_tests PRIVATE explorer_core m)
  target_compile_options(servo_bus_tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME servo_bus COMMAND servo_bus_tests)
  add_executable(sensor_tests tests/test_sensors.c src/sensors.c)
  target_include_directories(sensor_tests BEFORE PRIVATE tests/sensor_mock vendor/board)
  target_link_libraries(sensor_tests PRIVATE explorer_core m)
  target_compile_options(sensor_tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME sensors COMMAND sensor_tests)
  add_executable(uart_io_tests tests/test_uart_io.c src/uart_io.c)
  target_include_directories(uart_io_tests BEFORE PRIVATE tests/uart_mock)
  target_link_libraries(uart_io_tests PRIVATE explorer_core m)
  target_compile_options(uart_io_tests PRIVATE -Wall -Wextra -Werror)
  add_test(NAME uart_io COMMAND uart_io_tests)
else()
  add_compile_options(-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=firmware -fdebug-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=build-stm32)
  target_compile_options(explorer_core PRIVATE -ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=firmware -fdebug-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=build-stm32)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_board.py ${CMAKE_CURRENT_BINARY_DIR}/generated COMMAND_ERROR_IS_FATAL ANY)
  set(CPU -mcpu=cortex-m7 -mthumb -mfpu=fpv5-d16 -mfloat-abi=hard)
  target_compile_options(explorer_core PRIVATE ${CPU} -Os -ffunction-sections -fdata-sections)
  set(HAL vendor/Drivers/STM32H7xx_HAL_Driver)
  set(HAL_MODULES cortex dma dma_ex mdma exti flash flash_ex gpio hsem i2c i2c_ex pwr pwr_ex rcc rcc_ex tim tim_ex uart uart_ex spi spi_ex adc adc_ex rng rng_ex iwdg)
  set(HAL_SOURCES ${HAL}/Src/stm32h7xx_hal.c)
  foreach(name IN LISTS HAL_MODULES)
    list(APPEND HAL_SOURCES ${HAL}/Src/stm32h7xx_hal_${name}.c)
  endforeach()
  add_library(board_hal STATIC ${HAL_SOURCES} vendor/board/tim.c vendor/board/spi.c vendor/board/adc.c vendor/board/app_motor.c vendor/board/system_stm32h7xx.c ${CMAKE_CURRENT_BINARY_DIR}/generated/clock.c)
  target_include_directories(board_hal PUBLIC include ${CMAKE_CURRENT_BINARY_DIR}/generated vendor/board ${HAL}/Inc vendor/Drivers/CMSIS/Include vendor/Drivers/CMSIS/Device/ST/STM32H7xx/Include)
  target_compile_definitions(board_hal PUBLIC USE_HAL_DRIVER STM32H743xx)
  target_compile_options(board_hal PUBLIC ${CPU} PRIVATE -Os -ffunction-sections -fdata-sections)
  add_executable(explorer.elf src/main.c src/board.c src/uart_io.c src/sensors.c src/servo_bus.c vendor/board/startup_stm32h743vgtx.s)
  target_link_libraries(explorer.elf PRIVATE explorer_core board_hal m)
  target_compile_options(explorer.elf PRIVATE -Wall -Wextra -Werror -Os -ffunction-sections -fdata-sections -fno-fast-math)
  target_link_options(explorer.elf PRIVATE ${CPU} -nostartfiles -T${CMAKE_CURRENT_SOURCE_DIR}/target.ld --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/explorer.map -Wl,--print-memory-usage -Wl,--build-id=none)
  add_custom_command(TARGET explorer.elf POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:explorer.elf> ${CMAKE_CURRENT_BINARY_DIR}/explorer.bin
    COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:explorer.elf> ${CMAKE_CURRENT_BINARY_DIR}/explorer.hex
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:explorer.elf>)
endif()
